================================================================================
常用命令
================================================================================

# 验证模型
python test_camera_movement.py

# 查看可训练参数
python -c "
from camera_movement_rl import build_model_from_checkpoint
model, _ = build_model_from_checkpoint('train_records/coordinate_no_cam_mindcube/step_550')
model.print_trainable_parameters()
"

# 训练 (框架已准备，需实现数据加载)
python train_camera_movement.py \
    --adapter_path train_records/coordinate_no_cam_mindcube/step_550 \
    --output_dir checkpoints/camera_movement_rl \
    --num_epochs 10 \
    --learning_rate 1e-4

# 检查模型大小
python -c "
import torch
from camera_movement_rl import build_model_from_checkpoint
model, _ = build_model_from_checkpoint('train_records/coordinate_no_cam_mindcube/step_550')
total = sum(p.numel() for p in model.parameters())
trainable = sum(p.numel() for p in model.parameters() if p.requires_grad)
print(f'Total: {total/1e9:.2f}B, Trainable: {trainable/1e6:.2f}M')
"
